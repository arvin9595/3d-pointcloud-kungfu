        // 點雲粒子類別

        class CloudPoint {

            constructor() {

                this.boneIndex = 0;

                this.u = 0; // 骨骼線性插值 (0~1)

                this.theta = 0; // 圓柱體角度

                this.rRatio = 0; // 圓柱體半徑比例

                this.x = 0; this.y = 0; this.z = 0;

                this.vx = 0; this.vy = 0; this.vz = 0;

                this.baseBrightness = 0.3 + Math.random() * 0.7;

                this.reset();

            }

 

            reset() {

                this.boneIndex = Math.floor(Math.random() * BONES.length);

                this.u = Math.random();

                this.theta = Math.random() * Math.PI * 2;

                // 讓粒子分佈在骨骼周圍圓柱體內部與表面

                this.rRatio = Math.sqrt(Math.random()); 

                this.vx = 0; this.vy = 0; this.vz = 0;

            }

 

            update(bonesJoints) {

                const bone = BONES[this.boneIndex];

                const jA = bonesJoints[bone.from];

                const jB = bonesJoints[bone.to];

 

                // 計算骨骼軸向中心點

                const cx = jA.x + (jB.x - jA.x) * this.u;

                const cy = jA.y + (jB.y - jA.y) * this.u;

                const cz = jA.z + (jB.z - jA.z) * this.u;

 

                // 構建法向量垂直面以生成圓柱體點雲

                const dx = jB.x - jA.x;

                const dy = jB.y - jA.y;

                const dz = jB.z - jA.z;

                

                const r = bone.radius * this.rRatio;

                const jitter = (Math.random() - 0.5) * params.jitter * 3;

 

                // 簡單的三維切向座標映射

                const px = cx + Math.cos(this.theta) * r + jitter;

                const py = cy + Math.sin(this.theta) * r + jitter;

                const pz = cz + Math.sin(this.theta * 2) * (r * 0.5) + jitter;

 

                // 衝擊波擴散影響

                if (shockwave.active) {

                    const dX = px - shockwave.x;

                    const dY = py - shockwave.y;

                    const dZ = pz - shockwave.z;

                    const dist = Math.sqrt(dX*dX + dY*dY + dZ*dZ);

 

                    if (dist < shockwave.radius + 40 && dist > shockwave.radius - 40) {

                        const force = (1 - dist / 600) * 15;

                        this.vx += (dX / dist) * force;

                        this.vy += (dY / dist) * force;

                        this.vz += (dZ / dist) * force;

                    }

                }

 

                // 速度阻尼與回歸

                this.vx *= 0.88;

                this.vy *= 0.88;

                this.vz *= 0.88;

 

                this.x = px + this.vx;

                this.y = py + this.vy;

                this.z = pz + this.vz;

            }

        }

        // 3D 座標投影至 2D 畫布 (帶 RotateY, RotateX)

        function project3D(x, y, z) {

            // Y 軸旋轉

            let cosY = Math.cos(rotY), sinY = Math.sin(rotY);

            let x1 = x * cosY - z * sinY;

            let z1 = z * cosY + x * sinY;

 

            // X 軸旋轉

            let cosX = Math.cos(rotX), sinX = Math.sin(rotX);

            let y2 = y * cosX - z1 * sinX;

            let z2 = z1 * cosX + y * sinX;

 

            // 透視投影

            const fov = 450;

            const distance = 500;

            const scale = fov / (fov + z2 + distance);

 

            return {

                x: width / 2 + x1 * scale,

                y: height / 2 + 30 + y2 * scale,

                scale: scale,

                z: z2

            };

        }

 

        // 渲染主迴圈

        let lastTime = performance.now();

        let frameCount = 0;

 

        function animate(now) {

            // FPS 計算

            frameCount++;

            if (now - lastTime >= 1000) {

                document.getElementById('fps').innerText = `${frameCount} FPS`;

                frameCount = 0;

                lastTime = now;

            }

 

            // 清空畫布 (深色極簡背景)

            ctx.fillStyle = '#030305';

            ctx.fillRect(0, 0, width, height);

 

            // 繪製背景幾何網格 (網格點雲風格)

            ctx.fillStyle = 'rgba(255, 255, 255, 0.03)';

            for (let gx = -600; gx <= 600; gx += 120) {

                for (let gz = -600; gz <= 600; gz += 120) {

                    let p = project3D(gx, 180, gz);

                    ctx.fillRect(p.x, p.y, 1.5 * p.scale, 1.5 * p.scale);

                }

            }

 

            // 更新骨架運動

            updateSkeleton(now * 0.001);

            if (shockwave.active) {

                shockwave.radius += 12;

            }

            const renderList = [];

 

            for (let i = 0; i < particles.length; i++) {

                const p = particles[i];

                p.update(joints);

 

                const proj = project3D(p.x, p.y, p.z);

                renderList.push({

                    x: proj.x,

                    y: proj.y,

                    scale: proj.scale,

                    z: proj.z,

                    brightness: p.baseBrightness

                });

            }

 

            // 依據 Z-Buffer (深度) 排序，確保正確遮蔽與景深關係

            renderList.sort((a, b) => b.z - a.z);

 

            // 繪製灰階點雲 (Monochrome Rendering)

            for (let i = 0; i < renderList.length; i++) {

                const p = renderList[i];

 

                // 景深明暗計算 (越近越亮，越遠越暗)

                const depthAlpha = Math.min(1, Math.max(0.1, (p.z + 300) / 600));

                const grayValue = Math.floor(p.brightness * depthAlpha * 255);

                

                const renderSize = params.pointSize * p.scale;

 

                ctx.fillStyle = `rgb(${grayValue}, ${grayValue}, ${grayValue})`;

                

                // 畫出方形高科技點陣 (Point Grid Aspect)

                ctx.fillRect(p.x - renderSize / 2, p.y - renderSize / 2, renderSize, renderSize);

            }

 

            requestAnimationFrame(animate);

        }

 

        // 事件監聽 (UI 與 拖曳)

        window.addEventListener('resize', () => {

            width = canvas.width = window.innerWidth;

            height = canvas.height = window.innerHeight;

        });

 

        // 3D 視角旋轉控制

        canvas.addEventListener('mousedown', (e) => {

            isDragging = true;

            lastMouseX = e.clientX;

            lastMouseY = e.clientY;

        });

 

        window.addEventListener('mousemove', (e) => {

            if (isDragging) {

                const deltaX = e.clientX - lastMouseX;

                const deltaY = e.clientY - lastMouseY;

                rotY += deltaX * 0.005;

                rotX += deltaY * 0.005;

                lastMouseX = e.clientX;

                lastMouseY = e.clientY;

            }

        });

 

        window.addEventListener('mouseup', () => isDragging = false);

 

        // 觸控支援

        canvas.addEventListener('touchstart', (e) => {

            if (e.touches.length === 1) {

                isDragging = true;

                lastMouseX = e.touches[0].clientX;

                lastMouseY = e.touches[0].clientY;

            }

        });

 

        canvas.addEventListener('touchmove', (e) => {

            if (isDragging && e.touches.length === 1) {

                const deltaX = e.touches[0].clientX - lastMouseX;

                const deltaY = e.touches[0].clientY - lastMouseY;

                rotY += deltaX * 0.005;

                rotX += deltaY * 0.005;

                lastMouseX = e.touches[0].clientX;

                lastMouseY = e.touches[0].clientY;

            }

        });

 

        canvas.addEventListener('touchend', () => isDragging = false);

        canvas.addEventListener('click', (e) => {

            // 如果剛才是在拖曳視角，就不觸發衝擊波

            shockwave.x = 0;

            shockwave.y = 0;

            shockwave.z = 0;

            shockwave.radius = 0;

            shockwave.active = true;

        });

        document.getElementById('input-speed').addEventListener('input', (e) => {

            params.speed = parseFloat(e.target.value);

            document.getElementById('val-speed').innerText = params.speed.toFixed(1);

        });

 

        document.getElementById('input-jitter').addEventListener('input', (e) => {

            params.jitter = parseFloat(e.target.value);

            document.getElementById('val-jitter').innerText = params.jitter.toFixed(1);

        });

        const buttons = {
